    public static async void Run(IServiceProvider sp, MainPage page)
    {
        if (Mode is not { } mode) return;
        var theme = Environment.GetEnvironmentVariable("EAGLEEYE_UI_THEME");
        if (theme is not null) Application.Current!.UserAppTheme = theme == "dark" ? AppTheme.Dark : AppTheme.Light;
        var main = sp.GetRequiredService<MainViewModel>();
        await Task.Delay(1500);
        if (Environment.GetEnvironmentVariable("EAGLEEYE_UI_ACCENT") is { } accent)
        {
            var rgb = Convert.FromHexString(accent);
            var (bg, fg) = HeadingBarPalette.For(new RgbColor(rgb[0], rgb[1], rgb[2]));
            Application.Current!.Resources["HeadingBarBackgroundColor"] = Color.FromRgb(bg.R, bg.G, bg.B);
            Application.Current!.Resources["HeadingBarTextColor"] = Color.FromRgb(fg.R, fg.G, fg.B);
            Application.Current!.Resources["HeadingBarBrush"] = new SolidColorBrush(Color.FromRgb(bg.R, bg.G, bg.B));
        }

        var key = mode.StartsWith("settings") ? MainViewModel.SettingsKey : mode.StartsWith("rules") ? MainViewModel.RulesKey : mode.StartsWith("reports") ? MainViewModel.ReportsKey : MainViewModel.AppGroupsKey;
        main.SelectedItem = main.MenuItems.First(i => i.Key == key);
        Log($"page {key}");
        if (key != MainViewModel.AppGroupsKey) return;
        var vm = sp.GetRequiredService<AppGroupsViewModel>();
        await Task.Delay(1500);
        CheckFlyout(page);
        if (mode == "groups-drag")
        {
            Log("step A: highlight School"); vm.BeginDrag(vm.Sections[2].Apps[2]); vm.DragOver(vm.Sections[1]);
            await Task.Delay(5000);
            Log("step B: highlight Not in any group"); vm.DragOver(vm.Sections[2]);
            await Task.Delay(4000);
            Log("step C: end drag (no highlight)"); vm.EndDrag();
            await Task.Delay(4000);
            Log("step D: drop Paint on School"); vm.BeginDrag(vm.Sections[2].Apps[2]); vm.DragOver(vm.Sections[1]); await vm.DropAsync(vm.Sections[1]);
            await Task.Delay(4000);
            Log("step E: move-to sheet"); vm.Sections[0].Apps[0].MoveCommand.Execute(null);
        }
        else if (mode == "groups-rename")
        {
            vm.Sections[0].RenameCommand.Execute(null);
        }
        else if (mode == "groups-add")
        {
            vm.AddGroupCommand.Execute(null);
        }
        else if (mode == "groups-delete")
        {
            vm.Sections[0].DeleteCommand.Execute(null);
        }
    }

    private static void CheckFlyout(Element root)
    {
        foreach (var grid in root.GetVisualTreeDescendants().OfType<Grid>())
        {
            if (FlyoutBase.GetContextFlyout(grid) is MenuFlyout flyout && flyout.FirstOrDefault() is MenuFlyoutItem item)
            {
                Log($"flyout on {(grid.BindingContext as GroupableAppViewModel)?.DisplayName}: item text '{item.Text}', item context {item.BindingContext?.GetType().Name ?? "null"}, command {(item.Command is null ? "null" : "set")}");
                return;
            }
        }

        Log("no flyout found");
    }

        private long _revision = 1;
        public event Action? Changed;
        public AccountsLoadState LoadState => AccountsLoadState.Ready;
        public AppGroupsDto? Snapshot => new(_revision, null, _groups, _apps);
        public Task<bool> CreateAsync(string name) { _groups = [.. _groups, new(_next++, name)]; return Done(); }
        public Task<bool> RenameAsync(long groupId, string name) { _groups = [.. _groups.Select(g => g.GroupId == groupId ? g with { Name = name } : g)]; return Done(); }
        public Task<bool> DeleteAsync(long groupId) { _groups = [.. _groups.Where(g => g.GroupId != groupId)]; _apps = [.. _apps.Select(a => a.GroupId == groupId ? a with { GroupId = null } : a)]; return Done(); }
        public Task<bool> MoveAsync(string programPath, long? groupId) { Log($"move {programPath} -> {groupId}"); _apps = [.. _apps.Select(a => a.ProgramPath == programPath ? a with { GroupId = groupId } : a)]; return Done(); }
        private Task<bool> Done() { _revision++; Changed?.Invoke(); return Task.FromResult(true); }
    }
